import os,sys,json,hashlib,subprocess,datetime
from pathlib import Path
import yamtbx
from yamtbx.dataproc.dials.command_line import run_dials_auto
root=Path(__file__).resolve().parent
source=Path('/home/kuntaro/Data/images')
params=dict(filename_template=str(source/'cry00_??????.h5'),prefix=str(source/'cry00'),nr_range=(1,100),wdir=str(root),known_xs=None,overrides={},scan_varying=False,nproc=4)
manifest={'started':datetime.datetime.now().astimezone().isoformat(),'python':sys.executable,'yamtbx':yamtbx.__file__,'parameters':params,'git_commit':subprocess.check_output(['git','-C','/home/kuntaro/kamodev/kamo-merge-review','rev-parse','HEAD'],text=True).strip(),'git_status':subprocess.check_output(['git','-C','/home/kuntaro/kamodev/kamo-merge-review','status','--porcelain'],text=True),'input_sha256':{}}
for name in ['cry00_master.h5','cry00_data_000001.h5']:
 h=hashlib.sha256()
 with (source/name).open('rb') as f:
  for block in iter(lambda:f.read(1048576),b''):h.update(block)
 manifest['input_sha256'][name]=h.hexdigest()
(root/'manifest.json').write_text(json.dumps(manifest,indent=2))
run_dials_auto.run_dials_sequence(**params)
assert (root/'kamo_dials.pkl').is_file(), 'KAMO completion artifact missing'
print('KAMO sequence completed; scientific validation still required')
